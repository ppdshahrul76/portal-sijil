// 1. FUNGSI SIMPAN KEHADIRAN / PENDAFTARAN
function prosesPendaftaran(data) {
  return simpanDataKehadiran(data);
}

function simpanDataKehadiran(data) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName("Data") || ss.getSheetByName("KEHADIRAN") || ss.getActiveSheet();
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "No. Sijil", "Nama Penuh", "No. Kad Pengenalan", 
        "Nama Institusi", "Nama Program", "Penyertaan", "Anjuran Sektor", 
        "Tarikh Mula", "Tarikh Akhir", "Lokasi", "Alamat Emel", "Status Sijil"
      ]);
    }
    
    var timestamp = new Date();
    var jumlahBaris = sheet.getLastRow();
    var noSijilSiri = 1; 
    
    if (jumlahBaris > 1) {
      var dataSijil = sheet.getRange(2, 2, jumlahBaris - 1, 1).getValues(); 
      var siriTerbesar = 0;
      
      for (var i = 0; i < dataSijil.length; i++) {
        var stringSijil = dataSijil[i][0] ? dataSijil[i][0].toString() : ""; 
        if (stringSijil.includes("-")) {
          var nomborSahaja = parseInt(stringSijil.split("-")[1], 10); 
          if (!isNaN(nomborSahaja) && nomborSahaja > siriTerbesar) {
            siriTerbesar = nomborSahaja;
          }
        }
      }
      noSijilSiri = siriTerbesar + 1; 
    }
    
    var noSijilFormat = "PPDHS-" + String(noSijilSiri).padStart(5, '0');
    var tarikhMulaVal = data.tarikhMula || "";
    var tarikhAkhirVal = data.tarikhTamat || data.tarikhAkhir || "";
    
    sheet.appendRow([
      timestamp,
      noSijilFormat,
      (data.nama || "").toUpperCase().trim(),
      "'" + (data.ic || "").toString().trim(), 
      data.institusi || "",
      (data.program || "").toUpperCase().trim(),
      data.penyertaan || "PESERTA",
      data.anjuran || "",
      tarikhMulaVal,
      tarikhAkhirVal,
      (data.lokasi || "").toUpperCase().trim(),
      (data.emel || "").trim(),
      "BELUM DIJANA"
    ]);
    
    SpreadsheetApp.flush();
    
    return {
      status: "Sukses",
      SUCCESS: true,
      noSijil: noSijilFormat,
      message: "Data berjaya disimpan."
    };
    
  } catch (error) {
    throw new Error("Gagal menyimpan pendaftaran: " + error.toString());
  }
}

// 2. FASA ADMIN: JANA & HANTAR PUKAL (MENGGUNAKAN SCRIPT PROPERTIES SUPAYA SELAMAT DI GITHUB)
function hantarEmelPukalAdmin(idAdmin, passAdmin) {
  var props = PropertiesService.getScriptProperties();
  var VALID_ADMIN = props.getProperty('ADMIN_EMAIL') || "ppd-b050-cm10@moe-dl.edu.my";
  var VALID_PASS = props.getProperty('ADMIN_PASSWORD') || "DELIMa@54321";

  if (idAdmin.trim() !== VALID_ADMIN || passAdmin !== VALID_PASS) {
    return "Ralat: ID Admin atau Kata Laluan adalah salah!";
  }
  
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName("Data") || ss.getSheetByName("KEHADIRAN") || ss.getActiveSheet();
    
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return "Tiada data untuk diproses.";
    
    var headers = data[0].map(function(h) { 
      return h.toString().trim().toLowerCase().replace(/[\s\.\/]/g, ''); 
    });
    
    var idxNoSijil = headers.findIndex(function(h) { return h.includes("sijil"); });
    var idxNama = headers.findIndex(function(h) { return h.includes("nama") && !h.includes("institusi") && !h.includes("program"); });
    var idxIC = headers.findIndex(function(h) { return h.includes("kadpengenalan") || h.includes("ic"); });
    var idxInstitusi = headers.findIndex(function(h) { return h.includes("institusi"); });
    var idxProgram = headers.findIndex(function(h) { return h.includes("program"); });
    var idxPenyertaan = headers.findIndex(function(h) { return h.includes("penyertaan"); });
    var idxAnjuran = headers.findIndex(function(h) { return h.includes("anjuran") || h.includes("sektor"); });
    var idxTarikhMula = headers.findIndex(function(h) { return h.includes("mula") || (h.includes("tarikh") && !h.includes("akhir") && !h.includes("tamat")); });
    var idxTarikhTamat = headers.findIndex(function(h) { return h.includes("akhir") || h.includes("tamat"); });
    var idxLokasi = headers.findIndex(function(h) { return h.includes("lokasi"); });
    var idxEmel = headers.findIndex(function(h) { return h.includes("emel") || h.includes("email"); });
    var idxStatus = headers.findIndex(function(h) { return h.includes("status"); });
    
    if (idxStatus === -1) {
      return "Ralat: Lajur 'Status Sijil' tidak ditemui di Google Sheet.";
    }
    
    var bilBerjaya = 0;
    
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var statusSemasa = row[idxStatus] ? row[idxStatus].toString().trim() : "BELUM DIJANA";
      
      if (statusSemasa === "BELUM DIJANA") {
        var paketData = {
          noSijil: idxNoSijil !== -1 ? row[idxNoSijil] : "PPDHS-00000",
          nama: idxNama !== -1 ? row[idxNama] : "NAMA TIADA",
          ic: idxIC !== -1 ? row[idxIC] : "-",
          institusi: idxInstitusi !== -1 ? row[idxInstitusi] : "PPD HULU SELANGOR",
          program: idxProgram !== -1 ? row[idxProgram] : "PROGRAM SEKTOR",
          penyertaan: idxPenyertaan !== -1 ? row[idxPenyertaan] : "PESERTA",
          anjuran: idxAnjuran !== -1 ? row[idxAnjuran] : "Sektor",
          tarikhMula: idxTarikhMula !== -1 ? row[idxTarikhMula] : "",
          tarikhTamat: idxTarikhTamat !== -1 ? row[idxTarikhTamat] : "",
          lokasi: idxLokasi !== -1 ? row[idxLokasi] : "PPDHS",
          emel: idxEmel !== -1 ? row[idxEmel] : ""
        };
        
        if (paketData.emel.toString().trim() !== "") {
          var prosesOk = laksanaJanaDanKirim(paketData);
          if (prosesOk) {
            sheet.getRange(i + 1, idxStatus + 1).setValue("DIJANA");
            SpreadsheetApp.flush();
            bilBerjaya++;
          }
        }
      }
    }
    
    return "Sukses: Sebanyak " + bilBerjaya + " sijil digital baharu berjaya dijana dan diemelkan kepada peserta!";
  } catch (error) {
    return "Ralat pemprosesan pukal: " + error.toString();
  }
}

// 3. FUNGSI JANA PDF SLIDES & PENGHANTARAN EMEL (TERMASUK FASILITATOR)
function laksanaJanaDanKirim(data) {
  var TEMPLATE_SIJIL_PENYERTAAN = "1FcbCAKMhp6lua8LXyF7CFtQlmcGQunEDXFOK3SmPEEA";
  var TEMPLATE_SIJIL_PENGHARGAAN = "1y-g83saBUAPbNT_qS7pYFelVRNGov-GoGmKnSjr0NL0";
  
  // Penceramah dan Fasilitator akan menggunakan Templat Penghargaan
  var templateId = (data.penyertaan === "PENCERAMAH" || data.penyertaan === "FASILITATOR") ? TEMPLATE_SIJIL_PENGHARGAAN : TEMPLATE_SIJIL_PENYERTAAN;
  
  try {
    var failTemplat = DriveApp.getFileById(templateId);
    var salinanFail = failTemplat.makeCopy("Sijil_" + data.penyertaan + "_" + data.nama);
    var presentasiSijil = SlidesApp.openById(salinanFail.getId());
    var slaid = presentasiSijil.getSlides()[0]; 
    
    var paparanTarikhSijil = janaJulatTarikh(data.tarikhMula, data.tarikhTamat);
    
    slaid.replaceAllText("{{Nombor Sijil}}", data.noSijil);
    slaid.replaceAllText("{{Nama Penuh}}", data.nama.toUpperCase());
    slaid.replaceAllText("{{No. Kad Pengenalan}}", data.ic || "-");
    slaid.replaceAllText("{{Nama Institusi}}", data.institusi); 
    slaid.replaceAllText("{{Nama Program}}", data.program);
    slaid.replaceAllText("{{Penyertaan}}", data.penyertaan);
    slaid.replaceAllText("{{Anjuran}}", data.anjuran);
    slaid.replaceAllText("{{Tarikh Program}}", paparanTarikhSijil); 
    slaid.replaceAllText("{{Lokasi}}", data.lokasi);
    
    presentasiSijil.saveAndClose();
    
    var pdfBlob = salinanFail.getAs(MimeType.PDF);
    pdfBlob.setName("Sijil_" + data.noSijil + "_" + data.nama.replace(/\s+/g, '_') + ".pdf");
    
    var subjek = "SIJIL DIGITAL (" + data.penyertaan + "): " + data.program + " [" + data.noSijil + "]";
    var mesejEmel = "Assalamualaikum dan Salam Sejahtera Tuan/Puan,\n\nTahniah! Sijil digital dilampirkan.\n\nPejabat Pendidikan Daerah Hulu Selangor.";
                  
    MailApp.sendEmail({
      to: data.emel,
      subject: subjek,
      body: mesejEmel,
      attachments: [pdfBlob]
    });
    
    salinanFail.setTrashed(true); 
    return true;
  } catch(e) {
    return false;
  }
}

// 4. DAPATKAN SENARAI UNTUK JADUAL / DASHBOARD
function dapatkanSenaraiKehadiran() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName("Data") || ss.getSheetByName("KEHADIRAN") || ss.getActiveSheet();
    
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return [];
    
    var headers = data[0].map(function(h) { 
      return h.toString().trim().toLowerCase().replace(/[\s\.\/]/g, ''); 
    });
    
    var idxNoSijil = headers.findIndex(function(h) { return h.includes("sijil"); });
    var idxNama = headers.findIndex(function(h) { return h.includes("nama") && !h.includes("institusi") && !h.includes("program"); });
    var idxInstitusi = headers.findIndex(function(h) { return h.includes("institusi"); });
    var idxProgram = headers.findIndex(function(h) { return h.includes("program"); });
    var idxPenyertaan = headers.findIndex(function(h) { return h.includes("penyertaan"); });
    var idxAnjuran = headers.findIndex(function(h) { return h.includes("anjuran") || h.includes("sektor"); });
    var idxTarikhMula = headers.findIndex(function(h) { return h.includes("mula") || h.includes("tarikh"); });
    var idxStatus = headers.findIndex(function(h) { return h.includes("status"); });
    
    var senarai = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var tarikhMentah = idxTarikhMula !== -1 ? row[idxTarikhMula].toString() : "";
      var tahun = "2026";
      
      if (tarikhMentah.includes("-")) {
        tahun = tarikhMentah.split("-")[0];
      } else if (tarikhMentah.includes("/")) {
        var komponen = tarikhMentah.split("/");
        tahun = komponen[komponen.length - 1].split(" ")[0];
      }

      var valNoSijil = (idxNoSijil !== -1) ? row[idxNoSijil] : row[1];
      var valNama = (idxNama !== -1) ? row[idxNama] : row[2];
      var valInstitusi = (idxInstitusi !== -1) ? row[idxInstitusi] : row[4];
      var valProgram = (idxProgram !== -1) ? row[idxProgram] : row[5];
      var valPenyertaan = (idxPenyertaan !== -1) ? row[idxPenyertaan] : row[6];
      var valAnjuran = (idxAnjuran !== -1) ? row[idxAnjuran] : row[7]; 
      var valStatus = (idxStatus !== -1) ? row[idxStatus] : "BELUM DIJANA";

      senarai.push({
        noSijil: valNoSijil ? valNoSijil.toString().trim() : "",
        nama: valNama ? valNama.toString().trim() : "",
        institusi: valInstitusi ? valInstitusi.toString().trim() : "",
        program: valProgram ? valProgram.toString().trim() : "",
        penyertaan: valPenyertaan ? valPenyertaan.toString().trim() : "PESERTA",
        anjuran: valAnjuran ? valAnjuran.toString().trim() : "Tiada Sektor", 
        statusEmel: valStatus ? valStatus.toString().trim() : "BELUM DIJANA",
        tarikh: tarikhMentah,
        tahun: tahun
      });
    }
    return senarai.reverse(); 
  } catch (error) {
    return [];
  }
}

function padamRekodKehadiran(noSijil) {
  return padamRekodPangkalanData(noSijil);
}

function padamRekodPangkalanData(noSijil) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName("Data") || ss.getSheetByName("KEHADIRAN") || ss.getActiveSheet();
    
    var data = sheet.getDataRange().getValues();
    var headers = data[0].map(function(h) { 
      return h.toString().trim().toLowerCase().replace(/[\s\.]/g, ''); 
    });
    var idxNoSijil = headers.findIndex(function(h) { return h.includes("sijil"); });
    var targetIndex = idxNoSijil !== -1 ? idxNoSijil : 1;
    
    for (var i = 1; i < data.length; i++) {
      if (data[i][targetIndex].toString().trim() === noSijil.toString().trim()) {
        sheet.deleteRow(i + 1); 
        return "Berjaya";
      }
    }
    return "Ralat: Rekod nombor sijil tidak ditemui.";
  } catch (error) {
    return "Ralat: " + error.toString();
  }
}

// FUNGSI BANTUAN TARIKH
function janaJulatTarikh(mula, tamat) {
  if (!mula) return "";
  var formatMula = formatTarikhKemasan(mula);
  if (!tamat || mula.toString().trim() === tamat.toString().trim()) {
    return formatMula;
  }
  var formatTamat = formatTarikhKemasan(tamat);
  return formatMula + " - " + formatTamat;
}

function formatTarikhKemasan(tarikhAsal) {
  if (!tarikhAsal) return "";
  if (tarikhAsal instanceof Date) {
    var hari = tarikhAsal.getDate().toString().padStart(2, '0');
    var bulan = (tarikhAsal.getMonth() + 1).toString().padStart(2, '0'); 
    var tahun = tarikhAsal.getFullYear();
    return hari + "/" + bulan + "/" + tahun;
  }
  
  var bersih = tarikhAsal.toString().split(" ")[0];
  if (bersih.includes("-")) {
    var p = bersih.split("-");
    if (p.length === 3) return p[2].padStart(2, '0') + "/" + p[1].padStart(2, '0') + "/" + p[0];
  }
  if (bersih.includes("/")) {
    var p = bersih.split("/");
    if (p.length === 3) return p[0].padStart(2, '0') + "/" + p[1].padStart(2, '0') + "/" + p[2];
  }
  return bersih;
}
